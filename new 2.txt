	public static class CharacterPatternMap(){
		private char oops;
		private String[] pattern;
		
		public CharacterPatternMap(char oops,String[] pattern){
			this.oops=oops;
			this.pattern=pattern;
		}
		public char getChar(){
			return oops;
		}
		public String[] getPattern(){
			return pattern;
		}
	}
